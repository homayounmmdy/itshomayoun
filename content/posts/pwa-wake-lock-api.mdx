# 🔆 PWA: Wake Lock API - جلوگیری از خاموش شدن صفحه

**💡 مفهوم کلیدی**  
Wake Lock API به PWA اجازه میده **صفحه رو روشن نگه داره** - حتی وقتی کاربر با دستگاه کار نمی‌کنه. این برای اپ‌هایی مثل پخش ویدیو طولانی، ارائه‌ها (presentation)، اپ‌های ورزشی، اسکنر بارکد، یا اپ‌های آشپزی (که کاربر دستش بنده) حیاتیه. بدون این API، کاربر مجبوره هر چند ثانیه صفحه رو لمس کنه تا خاموش نشه!

**⚠️ دام رایج**  
- فراموش کردن `navigator.wakeLock.request('screen')` نیاز به **user activation** داره (بدون کلیک کاربر خطا میده!).
- عدم مدیریت `release` event (وقتی مرورگر به دلایلی قفل رو آزاد می‌کنه).
- فراموش کردن آزاد کردن wake lock وقتی کاربر از صفحه خارج میشه (باتری رو هدر میده!).
- عدم بررسی `document.visibilityState` (وقتی تب مخفی میشه، wake lock آزاد میشه!).
- نادیده گرفتن `navigator.wakeLock` در مرورگرهای بدون پشتیبانی (مثل iOS Safari).
- استفاده بدون `try/catch` (ممکنه خطای `NotAllowedError` بده).
- عدم احترام به تنظیمات باتری کاربر.
- فراموش کردن cleanup در unmount کامپوننت.

**📏 وضعیت پشتیبانی مرورگرها**

| مرورگر       | پشتیبانی | محدودیت                                  |
|--------------|-----------|-------------------------------------------|
| Chrome       | ✅        | نسخه 84+ + فقط HTTPS + user activation   |
| Edge         | ✅        | نسخه 84+                                 |
| Opera        | ✅        | نسخه 70+                                 |
| Samsung      | ✅        | کامل                                     |
| Safari       | ❌        | **پشتیبانی نمیشه**                        |
| iOS Safari   | ❌        | پشتیبانی نمیشه                            |
| Firefox      | ❌        | **پشتیبانی نمیشه** (flag لازم)            |

> ⚠️ **نکته حیاتی**: این API فقط در مرورگرهای **Chromium-based** کار می‌کنه. در iOS Safari و Firefox پشتیبانی نمیشه. همیشه fallback بذار!

### ۱. هوک React برای Wake Lock

```jsx
// hooks/useWakeLock.js
import { useState, useCallback, useEffect, useRef } from 'react';

export function useWakeLock() {
  const [isSupported, setIsSupported] = useState(false);
  const [isActive, setIsActive] = useState(false);
  const [error, setError] = useState(null);
  const wakeLockRef = useRef(null);

  useEffect(() => {
    setIsSupported('wakeLock' in navigator);
  }, []);

  // ✅ فعال‌سازی wake lock
  const requestWakeLock = useCallback(async () => {
    if (!isSupported) {
      setError('Wake Lock API در این مرورگر پشتیبانی نمیشه');
      return false;
    }

    try {
      setError(null);
      
      // ✅ درخواست wake lock (نیاز به user activation داره)
      wakeLockRef.current = await navigator.wakeLock.request('screen');
      setIsActive(true);

      // ✅ گوش دادن به release (مرورگر ممکنه قفل رو آزاد کنه)
      wakeLockRef.current.addEventListener('release', () => {
        console.log('🔓 Wake Lock released');
        setIsActive(false);
      });

      console.log('✅ Wake Lock activated');
      return true;
    } catch (err) {
      if (err.name === 'NotAllowedError') {
        setError('دسترسی به Wake Lock رد شد. لطفاً اول یه جا کلیک کنید.');
      } else {
        setError(err.message);
      }
      console.error('Wake Lock error:', err);
      return false;
    }
  }, [isSupported]);

  // ✅ غیرفعال‌سازی wake lock
  const releaseWakeLock = useCallback(async () => {
    if (wakeLockRef.current) {
      try {
        await wakeLockRef.current.release();
        wakeLockRef.current = null;
        setIsActive(false);
        console.log('✅ Wake Lock manually released');
      } catch (err) {
        console.error('Failed to release wake lock:', err);
      }
    }
  }, []);

  // ✅ مدیریت visibility change (وقتی تب مخفی میشه، wake lock آزاد میشه)
  useEffect(() => {
    if (!isSupported) return;

    const handleVisibilityChange = async () => {
      if (document.visibilityState === 'visible' && wakeLockRef.current === null) {
        // ✅ اگه قبلاً فعال بود و تب دوباره visible شد، دوباره درخواست بده
        // (اینجا باید state قبلی رو ذخیره کنی)
      }
    };

    document.addEventListener('visibilitychange', handleVisibilityChange);

    return () => {
      document.removeEventListener('visibilitychange', handleVisibilityChange);
    };
  }, [isSupported]);

  // ✅ cleanup هنگام unmount
  useEffect(() => {
    return () => {
      if (wakeLockRef.current) {
        wakeLockRef.current.release();
      }
    };
  }, []);

  return {
    isSupported,
    isActive,
    error,
    requestWakeLock,
    releaseWakeLock,
  };
}
```

### ۲. هوک پیشرفته با auto-reacquire

```jsx
// hooks/useWakeLockAdvanced.js
import { useState, useCallback, useEffect, useRef } from 'react';

export function useWakeLockAdvanced() {
  const [isSupported, setIsSupported] = useState(false);
  const [isActive, setIsActive] = useState(false);
  const [shouldStayActive, setShouldStayActive] = useState(false);
  const [error, setError] = useState(null);
  const wakeLockRef = useRef(null);

  useEffect(() => {
    setIsSupported('wakeLock' in navigator);
  }, []);

  // ✅ فعال‌سازی
  const enable = useCallback(async () => {
    if (!isSupported) {
      setError('Wake Lock API پشتیبانی نمیشه');
      return false;
    }

    try {
      setError(null);
      wakeLockRef.current = await navigator.wakeLock.request('screen');
      setIsActive(true);
      setShouldStayActive(true);

      wakeLockRef.current.addEventListener('release', () => {
        setIsActive(false);
      });

      console.log('✅ Wake Lock enabled');
      return true;
    } catch (err) {
      setError(err.message);
      return false;
    }
  }, [isSupported]);

  // ✅ غیرفعال‌سازی
  const disable = useCallback(async () => {
    setShouldStayActive(false);
    if (wakeLockRef.current) {
      await wakeLockRef.current.release();
      wakeLockRef.current = null;
      setIsActive(false);
    }
  }, []);

  // ✅ مدیریت visibility change (auto-reacquire)
  useEffect(() => {
    if (!isSupported) return;

    const handleVisibilityChange = async () => {
      if (document.visibilityState === 'visible' && shouldStayActive && !wakeLockRef.current) {
        try {
          wakeLockRef.current = await navigator.wakeLock.request('screen');
          setIsActive(true);
          
          wakeLockRef.current.addEventListener('release', () => {
            setIsActive(false);
          });
          
          console.log('✅ Wake Lock re-acquired after tab focus');
        } catch (err) {
          console.error('Failed to re-acquire wake lock:', err);
        }
      }
    };

    document.addEventListener('visibilitychange', handleVisibilityChange);

    return () => {
      document.removeEventListener('visibilitychange', handleVisibilityChange);
    };
  }, [isSupported, shouldStayActive]);

  // ✅ cleanup
  useEffect(() => {
    return () => {
      if (wakeLockRef.current) {
        wakeLockRef.current.release();
      }
    };
  }, []);

  return {
    isSupported,
    isActive,
    error,
    enable,
    disable,
  };
}
```

### ۴. استفاده در پخش ویدیو

```jsx
// components/VideoPlayerWithWakeLock.jsx
import { useEffect, useRef } from 'react';
import { useWakeLockAdvanced } from '@/hooks/useWakeLockAdvanced';

export function VideoPlayerWithWakeLock({ src }) {
  const videoRef = useRef(null);
  const { isActive, enable, disable } = useWakeLockAdvanced();

  // ✅ فعال‌سازی wake lock وقتی ویدیو پخش میشه
  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;

    const handlePlay = () => {
      enable();
    };

    const handlePause = () => {
      disable();
    };

    const handleEnded = () => {
      disable();
    };

    video.addEventListener('play', handlePlay);
    video.addEventListener('pause', handlePause);
    video.addEventListener('ended', handleEnded);

    return () => {
      video.removeEventListener('play', handlePlay);
      video.removeEventListener('pause', handlePause);
      video.removeEventListener('ended', handleEnded);
      disable();
    };
  }, [enable, disable]);

  return (
    <div className="space-y-3">
      <video
        ref={videoRef}
        src={src}
        controls
        className="w-full rounded-lg"
      />
      
      {isActive && (
        <div className="flex items-center gap-2 text-sm text-orange-600">
          <div className="w-2 h-2 bg-orange-500 rounded-full animate-pulse" />
          <span>صفحه روشن می‌مونه تا ویدیو در حال پخش هست</span>
        </div>
      )}
    </div>
  );
}
```

```jsx
// components/WorkoutTracker.jsx
import { useEffect } from 'react';
import { useWakeLockAdvanced } from '@/hooks/useWakeLockAdvanced';

export function WorkoutTracker() {
  const { isActive, enable, disable } = useWakeLockAdvanced();
  const [isRunning, setIsRunning] = useState(false);
  const [elapsedTime, setElapsedTime] = useState(0);

  // ✅ فعال‌سازی wake lock وقتی ورزش شروع میشه
  useEffect(() => {
    if (isRunning) {
      enable();
    } else {
      disable();
    }

    return () => disable();
  }, [isRunning, enable, disable]);

  // ✅ تایمر
  useEffect(() => {
    if (!isRunning) return;

    const interval = setInterval(() => {
      setElapsedTime((prev) => prev + 1);
    }, 1000);

    return () => clearInterval(interval);
  }, [isRunning]);

  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  return (
    <div className="max-w-md mx-auto p-6 bg-white rounded-xl shadow-lg space-y-6">
      <h2 className="text-2xl font-bold text-center">🏃 ورزش</h2>

      {/* ✅ تایمر */}
      <div className="text-center">
        <p className="text-6xl font-mono font-bold text-blue-600">
          {formatTime(elapsedTime)}
        </p>
      </div>

      {/* ✅ دکمه شروع/توقف */}
      <button
        onClick={() => setIsRunning(!isRunning)}
        className={`w-full px-6 py-4 rounded-lg font-medium text-lg transition-colors ${
          isRunning
            ? 'bg-red-600 text-white hover:bg-red-700'
            : 'bg-green-600 text-white hover:bg-green-700'
        }`}
      >
        {isRunning ? '⏸️ توقف' : '▶️ شروع'}
      </button>

      {/* ✅ وضعیت Wake Lock */}
      {isActive && (
        <div className="p-3 bg-orange-50 border border-orange-200 rounded-lg text-center">
          <p className="text-sm text-orange-800">
            🔆 صفحه روشن می‌مونه تا ورزش تموم بشه
          </p>
        </div>
      )}
    </div>
  );
}
```

```jsx
// components/PresentationMode.jsx
import { useEffect } from 'react';
import { useWakeLockAdvanced } from '@/hooks/useWakeLockAdvanced';

export function PresentationMode({ slides }) {
  const { isActive, enable, disable } = useWakeLockAdvanced();
  const [currentSlide, setCurrentSlide] = useState(0);

  // ✅ فعال‌سازی wake lock در حالت ارائه
  useEffect(() => {
    enable();
    return () => disable();
  }, [enable, disable]);

  // ✅ ناوبری با کیبورد
  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'ArrowRight' || e.key === ' ') {
        setCurrentSlide((prev) => Math.min(prev + 1, slides.length - 1));
      } else if (e.key === 'ArrowLeft') {
        setCurrentSlide((prev) => Math.max(prev - 1, 0));
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [slides.length]);

  return (
    <div className="min-h-screen bg-gray-900 text-white flex flex-col">
      {/* ✅ Slide */}
      <div className="flex-1 flex items-center justify-center p-8">
        <div className="max-w-4xl text-center">
          <h1 className="text-5xl font-bold mb-4">
            {slides[currentSlide].title}
          </h1>
          <p className="text-xl text-gray-300">
            {slides[currentSlide].content}
          </p>
        </div>
      </div>

      {/* ✅ Navigation */}
      <div className="p-4 bg-gray-800 flex items-center justify-between">
        <button
          onClick={() => setCurrentSlide((prev) => Math.max(prev - 1, 0))}
          disabled={currentSlide === 0}
          className="px-4 py-2 bg-gray-700 rounded disabled:opacity-50"
        >
          ← قبلی
        </button>

        <span className="text-sm">
          {currentSlide + 1} / {slides.length}
        </span>

        <button
          onClick={() => setCurrentSlide((prev) => Math.min(prev + 1, slides.length - 1))}
          disabled={currentSlide === slides.length - 1}
          className="px-4 py-2 bg-gray-700 rounded disabled:opacity-50"
        >
          بعدی →
        </button>
      </div>

      {/* ✅ Wake Lock indicator */}
      {isActive && (
        <div className="fixed top-4 right-4 px-3 py-1 bg-orange-600 rounded-full text-xs">
          🔆 Wake Lock فعال
        </div>
      )}
    </div>
  );
}
```
