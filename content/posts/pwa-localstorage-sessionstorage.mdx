# 💾 PWA: LocalStorage و SessionStorage - ذخیره‌سازی ساده

**💡 مفهوم کلیدی**  
دو API ساده برای ذخیره‌سازی داده در مرورگر:
- **LocalStorage**: داده‌ها **دائمی** هستن (تا وقتی کاربر پاکشون نکنه) و بین همه‌ی تب‌های همون origin به اشتراک گذاشته میشن.
- **SessionStorage**: داده‌ها **موقتی** هستن (تا وقتی تب بسته بشه) و فقط در همون تب در دسترسن.

هر دو **فقط string** ذخیره می‌کنن، **sync** هستن (UI رو بلاک می‌کنن) و **۵-۱۰ مگابایت** ظرفیت دارن. برای تنظیمات کوچک، تم کاربر، token و داده‌های ساده مناسبن - ولی برای داده‌های حجیم از IndexedDB استفاده کن.

**⚠️ دام رایج**  
- ذخیره‌ی object بدون `JSON.stringify` → تبدیل به `"[object Object]"` میشه!
- فراموش کردن `JSON.parse` با try/catch → اگه داده خراب باشه، اپ کرش می‌کنه.
- عدم مدیریت `QuotaExceededError` → وقتی فضا پر میشه، اپ کرش می‌کنه.
- استفاده در SSR (Next.js) → `window is not defined` خطا میده.
- ذخیره‌ی داده‌های حساس (JWT، پسورد) → در برابر XSS آسیب‌پذیره.
- عدم گوش دادن به `storage` event → تغییرات در تب دیگه sync نمیشه.
- استفاده برای داده‌های حجیم (بیشتر از ۵MB) → خطای quota میده.
- عدم بررسی `null` هنگام خواندن → `Cannot read property of null`.

**📏 مقایسه LocalStorage vs SessionStorage**

| ویژگی              | LocalStorage          | SessionStorage        |
|--------------------|------------------------|------------------------|
| ماندگاری           | دائمی (تا پاک شدن)    | موقتی (تا بسته شدن تب) |
| ظرفیت              | ۵-۱۰ MB               | ۵-۱۰ MB               |
| اشتراک بین تب‌ها   | ✅ بله                 | ❌ خیر                 |
| نوع داده           | فقط string            | فقط string            |
| Sync/Async         | Sync ❌               | Sync ❌               |
| در Service Worker  | ✅ بله                 | ❌ خیر                 |
| مناسب برای         | تنظیمات، token، theme | داده‌ی فرم، wizard    |

**🛠️ راه‌حل سریع در کد**

### ۱. هوک امن و type-safe برای LocalStorage

```jsx
// hooks/useLocalStorage.js
import { useState, useCallback, useEffect } from 'react';

export function useLocalStorage(key, initialValue) {
  // ✅ مقدار اولیه: از localStorage بخون یا initialValue استفاده کن
  const [storedValue, setStoredValue] = useState(() => {
    if (typeof window === 'undefined') {
      return initialValue; // ✅ SSR safe
    }

    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.warn(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  // ✅ ذخیره‌ی مقدار جدید
  const setValue = useCallback((value) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      
      if (typeof window !== 'undefined') {
        window.localStorage.setItem(key, JSON.stringify(valueToStore));
      }
    } catch (error) {
      if (error.name === 'QuotaExceededError') {
        console.error('localStorage quota exceeded!');
        // ✅ می‌تونی اینجا داده‌های قدیمی رو پاک کنی
      }
      console.warn(`Error setting localStorage key "${key}":`, error);
    }
  }, [key, storedValue]);

  // ✅ حذف مقدار
  const removeValue = useCallback(() => {
    try {
      if (typeof window !== 'undefined') {
        window.localStorage.removeItem(key);
      }
      setStoredValue(initialValue);
    } catch (error) {
      console.warn(`Error removing localStorage key "${key}":`, error);
    }
  }, [key, initialValue]);

  // ✅ گوش دادن به تغییرات از تب دیگه
  useEffect(() => {
    if (typeof window === 'undefined') return;

    const handleStorageChange = (event) => {
      if (event.key === key && event.newValue !== null) {
        try {
          setStoredValue(JSON.parse(event.newValue));
        } catch (error) {
          console.warn(`Error parsing storage event for key "${key}":`, error);
        }
      } else if (event.key === key && event.newValue === null) {
        setStoredValue(initialValue);
      }
    };

    window.addEventListener('storage', handleStorageChange);
    return () => window.removeEventListener('storage', handleStorageChange);
  }, [key, initialValue]);

  return [storedValue, setValue, removeValue];
}

// ✅ استفاده:
function ThemeToggle() {
  const [theme, setTheme] = useLocalStorage('theme', 'light');

  return (
    <button
      onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}
      className="p-2 bg-gray-200 dark:bg-gray-800 rounded"
    >
      {theme === 'light' ? '🌙' : '☀️'}
    </button>
  );
}
```

export function useSessionStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    if (typeof window === 'undefined') {
      return initialValue;
    }

    try {
      const item = window.sessionStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {

      return initialValue;
    }
  });

  const setValue = useCallback((value) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      
      if (typeof window !== 'undefined') {
        window.sessionStorage.setItem(key, JSON.stringify(valueToStore));
      }
    } catch (error) {
      if (error.name === 'QuotaExceededError') {
        console.error('sessionStorage quota exceeded!');
      }
      console.warn(`Error setting sessionStorage key "${key}":`, error);
    }
  }, [key, storedValue]);

  const removeValue = useCallback(() => {
    try {
      if (typeof window !== 'undefined') {
        window.sessionStorage.removeItem(key);
      }
      setStoredValue(initialValue);
    } catch (error) {
      console.warn(`Error removing sessionStorage key "${key}":`, error);
    }
  }, [key, initialValue]);

  return [storedValue, setValue, removeValue];
}

// ✅ استفاده: ذخیره‌ی state فرم چندمرحله‌ای
function MultiStepForm() {

### ۳. Wrapper حرفه‌ای با fallback و encryption

```js
// lib/storage.js

// ✅ بررسی پشتیبانی
export const isLocalStorageSupported = () => {
  if (typeof window === 'undefined') return false;
  
  try {
    const testKey = '__storage_test__';
    window.localStorage.setItem(testKey, 'test');
    window.localStorage.removeItem(testKey);
    return true;
  } catch (e) {
    return false;
  }
};

// ✅ Fallback: in-memory storage
const memoryStorage = {
  store: new Map(),
  getItem(key) { return this.store.get(key) || null; },
  setItem(key, value) { this.store.set(key, value); },
  removeItem(key) { this.store.delete(key); },
  clear() { this.store.clear(); },
};

// ✅ انتخاب storage مناسب
export const getStorage = (type = 'local') => {
  if (!isLocalStorageSupported()) {
    console.warn('Storage not supported, using in-memory fallback');
    return memoryStorage;
  }
  
  return type === 'session' ? window.sessionStorage : window.localStorage;
};

// ✅ Safe get
export const storageGet = (key, defaultValue = null, type = 'local') => {
  try {
    const storage = getStorage(type);
    const item = storage.getItem(key);
    
    if (item === null) return defaultValue;
    
    // ✅ تلاش برای parse JSON
    try {
      return JSON.parse(item);
    } catch {
      return item; // اگه JSON نبود، همون string رو برگردون
    }
  } catch (error) {
    console.warn(`Storage get error for key "${key}":`, error);
    return defaultValue;
  }
};

// ✅ Safe set
export const storageSet = (key, value, type = 'local') => {
  try {
    const storage = getStorage(type);
    const serialized = JSON.stringify(value);
    storage.setItem(key, serialized);
    return true;
  } catch (error) {
    if (error.name === 'QuotaExceededError') {
      console.error('Storage quota exceeded!');
      
      // ✅ تلاش برای پاک کردن داده‌های قدیمی
      try {
        const storage = getStorage(type);
        const keys = Object.keys(storage);
        if (keys.length > 0) {
          storage.removeItem(keys[0]); // پاک کردن اولین key
          storage.setItem(key, JSON.stringify(value)); // تلاش دوباره
          return true;
        }
      } catch (retryError) {
        console.error('Retry failed:', retryError);
      }
    }
    return false;
  }
};

// ✅ Safe remove
export const storageRemove = (key, type = 'local') => {
  try {
    const storage = getStorage(type);
    storage.removeItem(key);
    return true;
  } catch (error) {
    console.warn(`Storage remove error for key "${key}":`, error);
    return false;
  }
};

// ✅ پاک کردن همه
export const storageClear = (type = 'local') => {
  try {
    const storage = getStorage(type);
    storage.clear();
    return true;
  } catch (error) {
    console.warn('Storage clear error:', error);
    return false;
  }
};

// ✅ بررسی فضای استفاده‌شده
export const getStorageUsage = (type = 'local') => {
  try {
    const storage = getStorage(type);
    let total = 0;
    
    for (const key in storage) {
      if (storage.hasOwnProperty(key)) {
        total += storage[key].length + key.length;
      }
    }
    
    return {
      bytes: total * 2, // هر کاراکتر UTF-16 = 2 bytes
      KB: ((total * 2) / 1024).toFixed(2),
      MB: ((total * 2) / 1024 / 1024).toFixed(2),
    };
  } catch (error) {
    return null;
  }
};
```

import { storageGet, storageSet, getStorageUsage } from '@/lib/storage';

export function UserPreferences() {

  useEffect(() => {

### ۵. Sync بین تب‌ها

```jsx
// hooks/useStorageSync.js
import { useEffect } from 'react';

export function useStorageSync(key, callback) {
  useEffect(() => {
    if (typeof window === 'undefined') return;

    const handleStorageChange = (event) => {
      if (event.key === key) {
        try {
          const newValue = event.newValue ? JSON.parse(event.newValue) : null;
          callback(newValue, event.oldValue ? JSON.parse(event.oldValue) : null);
        } catch (error) {
          console.warn('Storage sync error:', error);
        }
      }
    };

    window.addEventListener('storage', handleStorageChange);
    return () => window.removeEventListener('storage', handleStorageChange);
  }, [key, callback]);
}

// ✅ استفاده: sync کردن login state بین تب‌ها
function App() {
  useStorageSync('auth-token', (newToken, oldToken) => {
    if (newToken && !oldToken) {
      console.log('✅ User logged in in another tab');
      window.location.reload();
    } else if (!newToken && oldToken) {
      console.log('🚪 User logged out in another tab');
      window.location.href = '/login';
    }
  });

  return <div>...</div>;
}

// lib/secure-storage.js

// ✅ XOR encryption ساده (برای داده‌های غیرحیاتی)
const xorEncrypt = (text, key = 'secret-key') => {
  let result = '';
  for (let i = 0; i < text.length; i++) {
    result += String.fromCharCode(
      text.charCodeAt(i) ^ key.charCodeAt(i % key.length)
    );
  }
  return btoa(result); // base64 encode
};

const xorDecrypt = (encrypted, key = 'secret-key') => {
  const decoded = atob(encrypted);
  let result = '';
  for (let i = 0; i < decoded.length; i++) {
    result += String.fromCharCode(
      decoded.charCodeAt(i) ^ key.charCodeAt(i % key.length)
    );
  }
  return result;
};

// ✅ Secure storage
export const secureStorageSet = (key, value) => {
  try {
    const serialized = JSON.stringify(value);
    const encrypted = xorEncrypt(serialized);
    window.localStorage.setItem(key, encrypted);
    return true;
  } catch (error) {
    console.error('Secure storage set error:', error);
    return false;
  }
};

export const secureStorageGet = (key) => {
  try {
    const encrypted = window.localStorage.getItem(key);
    if (!encrypted) return null;
    
    const decrypted = xorDecrypt(encrypted);
    return JSON.parse(decrypted);
  } catch (error) {
    console.error('Secure storage get error:', error);
    return null;
  }
};

// ✅ استفاده:
secureStorageSet('user-token', 'abc123xyz');
const token = secureStorageGet('user-token'); // 'abc123xyz'
```

### ۷. پاک‌سازی خودکار داده‌های قدیمی

```js
// lib/storage-cleanup.js

export const cleanupOldStorage = (maxAgeMs = 7 * 24 * 60 * 60 * 1000) => {
  // ✅ پاک کردن داده‌های قدیمی‌تر از ۷ روز
  try {
    const now = Date.now();
    const keysToRemove = [];

    for (let i = 0; i < window.localStorage.length; i++) {
      const key = window.localStorage.key(i);
      
      if (key.startsWith('temp-') || key.startsWith('cache-')) {
        const item = window.localStorage.getItem(key);
        
        try {
          const parsed = JSON.parse(item);
          
          if (parsed.timestamp && (now - parsed.timestamp) > maxAgeMs) {
            keysToRemove.push(key);
          }
        } catch {
          // اگه JSON نبود، skip کن
        }
      }
    }

    keysToRemove.forEach((key) => window.localStorage.removeItem(key));
    
    console.log(`✅ Cleaned up ${keysToRemove.length} old storage items`);
    return keysToRemove.length;
  } catch (error) {
    console.error('Storage cleanup error:', error);
    return 0;
  }
};

// ✅ اجرا هنگام لود اپ
if (typeof window !== 'undefined') {
  cleanupOldStorage();
}
```
